// 미적용 초안. 기존 isAdmin() 함수 및 나머지 규칙은 유지합니다.
// 기존 match /{document=**}의 일괄 거부보다 앞에 아래 블록만 검토하여 추가합니다.
// 이 파일을 저장하는 행위는 운영 Firestore 권한을 바꾸지 않습니다.
// 새 쓰기 경로를 활성화하는 변경이므로 실제 규칙과 권한 범위를 확인한 배포가 필요합니다.

function validSiteEvent(d) {
  return d.keys().hasOnly([
      'schema','event','source','page','service','placement','target','test','at'
    ])
    && d.keys().hasAll([
      'schema','event','source','page','service','placement','target','test','at'
    ])
    && d.schema == 1
    && d.event in [
      'page_view','case_open','shop_open','quote_start','quote_result','contact_prepare',
      'kakao_open','phone_open','sms_open','signup_complete','diagnostic'
    ]
    && d.source in [
      'youtube','shorts','instagram','reels','threads','blog',
      'naver','google','profile','share','direct','other'
    ]
    && d.page in ['home','service','area','shop','other']
    && d.service in [
      'move-in','regular','special-home','office','clinic','post-build',
      'window','aircon','washer','mattress','carpet','disinfect','unknown'
    ]
    && d.placement in [
      'hero','sticky','quote','header','footer','service','other','test'
    ]
    && d.target in ['quote','direct','none','test']
    && d.test is bool
    && d.at == request.time
    && ((d.test == false && d.event != 'diagnostic') || (d.test == true && isAdmin()));
}

match /site_events/{eventId} {
  allow read: if isAdmin();
  allow create: if eventId.size() == 20
    && eventId.matches('^[A-Za-z0-9]+$')
    && validSiteEvent(request.resource.data);
  allow update, delete: if false;
}
