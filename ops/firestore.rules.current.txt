rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function owner(id) {
      return signedIn() && request.auth.uid == id;
    }

    function isAdmin() {
      return signedIn()
        && request.auth.uid == 'H1MfdqGkFRZzU8qkqwXTsiKv3CO2'
        && request.auth.token.firebase.sign_in_provider == 'google.com';
    }

    function validMember(d) {
      return d.keys().hasOnly([
          'name',
          'phone',
          'area',
          'free',
          'ad',
          'code',
          'ref',
          'joined',
          'status',
          'couponStatus',
          'freeStatus',
          'memo',
          'points',
          'grade',
          'ownerUid'
        ])
        && d.keys().hasAll([
          'name',
          'phone',
          'area',
          'free',
          'ad',
          'code',
          'ref',
          'joined',
          'status',
          'couponStatus',
          'freeStatus',
          'memo',
          'points',
          'grade',
          'ownerUid'
        ])
        && d.name is string
        && d.name.size() > 0
        && d.name.size() <= 30
        && d.phone is string
        && d.phone.matches('^01[016789]-?[0-9]{3,4}-?[0-9]{4}$')
        && d.area is string
        && d.area.size() <= 40
        && d.free is string
        && d.free.size() <= 80
        && d.ad is bool
        && d.code is string
        && d.code.matches('^OK-[A-Z2-9]{6}$')
        && d.ref is string
        && d.ref.size() <= 20
        && d.joined is timestamp
        && d.ownerUid is string
        && d.ownerUid.size() > 0
        && d.ownerUid.size() <= 128
        && d.status in [
          '신규',
          '상담중',
          '예약확정',
          '작업완료',
          '종료'
        ]
        && d.couponStatus in [
          '미사용',
          '사용',
          '취소'
        ]
        && d.freeStatus in [
          '미제공',
          '제공완료',
          '취소'
        ]
        && d.memo is string
        && d.memo.size() <= 1000
        && d.points is number
        && d.points >= 0
        && d.points <= 10000000
        && d.grade in [
          '새싹',
          '실버',
          '골드',
          'VIP'
        ];
    }

    function validReceipt(d, id) {
      return d.keys().hasAll([
          'ownerUid',
          'requestId',
          'code',
          'joined'
        ])
        && d.keys().hasOnly([
          'ownerUid',
          'requestId',
          'code',
          'joined'
        ])
        && d.ownerUid == id
        && d.requestId == id
        && id.size() <= 128
        && d.code is string
        && d.code.matches('^OK-[A-Z2-9]{6}$')
        && d.joined == request.time;
    }

    match /member_requests/{id} {

      // 회원정보 조회는 관리자만
      allow read: if isAdmin();

      // 회원가입 생성
      allow create: if owner(id)
        && validMember(request.resource.data)
        && request.resource.data.ownerUid == id
        && request.resource.data.joined == request.time
        && request.resource.data.status == '신규'
        && request.resource.data.couponStatus == '미사용'
        && request.resource.data.freeStatus == '미제공'
        && request.resource.data.memo == ''
        && request.resource.data.points == 0
        && request.resource.data.grade == '새싹'
        && getAfter(
          /databases/$(database)/documents/member_receipts/$(id)
        ).data.code == request.resource.data.code
        && getAfter(
          /databases/$(database)/documents/member_receipts/$(id)
        ).data.joined == request.time;

      // 운영 관리 필드만 관리자 수정 가능
      allow update: if isAdmin()
        && validMember(request.resource.data)
        && request.resource.data
          .diff(resource.data)
          .affectedKeys()
          .hasOnly([
            'status',
            'couponStatus',
            'freeStatus',
            'grade',
            'points',
            'memo'
          ]);

      // 관리자만 회원정보 삭제 가능
      allow delete: if isAdmin();
    }

    match /member_receipts/{id} {

      // 본인 또는 관리자만 단건 조회
      allow get: if owner(id) || isAdmin();

      // 전체 목록 조회 차단
      allow list: if false;

      // 본인 UID 영수증만 생성
      allow create: if owner(id)
        && validReceipt(request.resource.data, id);

      allow update: if false;

      // 관리자만 쿠폰 영수증 삭제 가능
      allow delete: if isAdmin();
    }

    // 추천제품
    // 고객은 상품 목록 읽기 가능
    // 추가·수정·삭제는 관리자만 가능
    match /products/{productId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    // 추천제품 클릭 통계
    // 고객은 클릭 기록 생성만 가능
    // 통계 조회는 관리자만 가능
    match /product_clicks/{clickId} {

      allow read: if isAdmin();

      allow create: if
        request.resource.data.keys().hasOnly([
          'product',
          'src',
          'at'
        ])
        && request.resource.data.keys().hasAll([
          'product',
          'src',
          'at'
        ])
        && request.resource.data.product is string
        && request.resource.data.product.size() > 0
        && request.resource.data.product.size() <= 128
        && request.resource.data.src is string
        && request.resource.data.src.size() <= 32
        && request.resource.data.at == request.time;

      allow update, delete: if false;
    }

    // 사이트 이용통계: 아래 열거값 8개와 서버 시각 at만 저장합니다.
    // 이름·연락처·상담 본문·사용자 ID 필드는 허용하지 않습니다.
    // 방문자는 운영 이벤트 생성만, 기존 Google 관리자는 조회·시험 생성이 가능합니다.
    // 수정은 항상 거부하며, 90일이 지난 통계만 기존 Google 관리자가 정리합니다.
    // 하위 컬렉션과 다른 컬렉션은 이 정리 권한에 포함되지 않습니다.
    function validSiteEvent(d) {
      return d.keys().hasOnly([
          'schema','event','source','page','service','placement','target','test','at'
        ])
        && d.keys().hasAll([
          'schema','event','source','page','service','placement','target','test','at'
        ])
        && d.schema is int
        && d.schema == 1
        && d.event in [
          'page_view','case_open','shop_open','quote_start','quote_result','contact_prepare',
          'kakao_open','phone_open','sms_open','signup_complete','diagnostic'
        ]
        && d.source in [
          'youtube','shorts','instagram','reels','threads','blog',
          'naver','google','profile','share','direct','other'
        ]
        && d.page in ['home','service','area','shop','other']
        && d['service'] in [
          'move-in','regular','special-home','office','clinic','post-build',
          'window','aircon','washer','mattress','carpet','disinfect','unknown'
        ]
        && d.placement in [
          'hero','sticky','quote','header','footer','service','other','test'
        ]
        && d.target in ['quote','direct','none','test']
        && d.test is bool
        && d.at is timestamp
        && d.at == request.time
        && ((d.test == false && d.event != 'diagnostic') || (d.test == true && isAdmin()));
    }

    function expiredSiteEventAt(at) {
      return at is timestamp
        && at < request.time - duration.value(90, 'd');
    }

    match /site_events/{eventId} {
      allow read: if isAdmin();
      allow create: if eventId.size() == 20
        && eventId.matches('^[A-Za-z0-9]+$')
        && validSiteEvent(request.resource.data);
      allow update: if false;
      allow delete: if isAdmin() && expiredSiteEventAt(resource.data.at);
    }

    // 나머지는 전부 차단
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
